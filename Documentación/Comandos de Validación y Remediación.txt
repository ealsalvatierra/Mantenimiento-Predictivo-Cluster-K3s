===============================================================================
GUIA DE COMANDOS DE VALIDACION, REMEDIACION Y DIAGNOSTICO - CLUSTER K3S
Proyecto: Sistema de Mantenimiento Predictivo Industrial (Namespace: predictivo)
===============================================================================

1. VALIDACIONES Y MONITOREO DEL CLUSTER
-------------------------------------------------------------------------------
# Estado de los nodos del cluster (Master, Worker 1 y Raspberry Pi):
sudo kubectl get nodes -o wide

# Estado detallado de todos los pods desplegados:
sudo kubectl get pods -n predictivo -o wide

# Monitoreo en tiempo real de eventos y cambios de estado en los pods:
sudo kubectl get pods -n predictivo -w

# Suscribirse en vivo al broker MQTT (Verificar envio desde la Raspberry Pi):
sudo kubectl exec -it -n predictivo deployment/mosquitto-broker -- mosquitto_sub -t "industrial/sensores/pi"


2. INSPECCION DE LOGS Y DIAGNOSTICO DE COMPONENTES
-------------------------------------------------------------------------------
# Logs de la IA / Sensor Edge en la Raspberry Pi:
sudo kubectl logs -n predictivo -l app=edge-sensor-classifier --tail=50 -f

# Logs del Persistidor Python (Mosquitto -> InfluxDB):
sudo kubectl logs -n predictivo -l app=mqtt-persister --tail=50 -f

# Logs del servidor visualizador Grafana:
sudo kubectl logs -n predictivo -l app=grafana --tail=50

# Logs de la base de datos InfluxDB:
sudo kubectl logs -n predictivo -l app=influxdb --tail=50


3. REMEDIACIONES Y RESTABLECIMIENTO DE SERVICIOS
-------------------------------------------------------------------------------
# Reiniciar la cadena completa de datos (Reestablecer sockets MQTT/InfluxDB):
sudo kubectl rollout restart deployment mqtt-persister edge-sensor-classifier -n predictivo

# Reiniciar Grafana:
sudo kubectl rollout restart deployment grafana -n predictivo

# Simular fallo critico / Prueba de Auto-Healing en el Broker MQTT:
sudo kubectl delete pod -n predictivo -l app=mosquitto

# Forzar la eliminacion de un pod trabado en estado 'Terminating':
sudo kubectl delete pod <NOMBRE_DEL_POD> -n predictivo --force --grace-period=0


4. VALIDACION DE PERSISTENCIA Y RED
-------------------------------------------------------------------------------
# Verificar la direccion IP actual de la maquina virtual (Master / Workers):
ip a

# Verificar los archivos persistentes de Grafana (Ejecutar en worker1-node):
ls -la /var/lib/grafana-data

# Corregir o asegurar permisos de lectura/escritura (Ejecutar en worker1-node):
sudo chmod -R 777 /var/lib/grafana-data
