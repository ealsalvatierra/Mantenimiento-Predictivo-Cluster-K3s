Paso 1: Cambiar el nombre de host
sudo hostnamectl set-hostname worker2-pi

Paso 2: Configurar la IP estática en Netplan
# Nota: Reemplaza 50-cloud-init.yaml por el nombre exacto de tu archivo si es distinto (usa 'ls /etc/netplan/' para ver cual es)
sudo cp /etc/netplan/50-cloud-init.yaml /etc/netplan/01-custom-network.yaml
sudo mv /etc/netplan/50-cloud-init.yaml /etc/netplan/50-cloud-init.yaml.disabled

- Editar el nuevo archivo:
sudo nano /etc/netplan/01-custom-network.yaml

# Reemplazar todo el contenido por esto:
network:
  version: 2
  ethernets:
    eth0:   # Verifica la interfaz con 'ip a' (suele ser eth0 o end0)
      dhcp4: no
      addresses:
        - 192.168.1.102/24   # IP del Worker 2 (Raspberry Pi)
      routes:
        - to: default
          via: 192.168.1.1   # IP del Gateway/Router de la red
      nameservers:
        addresses: [8.8.8.8, 1.1.1.1]

- Aplicar cambios de red:
sudo netplan apply

Paso 3: Preparar el sistema operativo para K3s
# Desactivar la memoria Swap
sudo swapoff -a
sudo sed -i '/ swap / s/^\(.*\)$/#\1/g' /etc/fstab
sudo systemctl disable --now dphys-swapfile 2>/dev/null || true

# Cargar módulos del kernel
sudo modprobe overlay
sudo modprobe br_netfilter

Paso 4: Probar la conectividad con el nodo Master
ping 192.168.1.100 #IP de master

Paso 5: Unir la Raspberry Pi al Clúster K3s
curl -sfL https://get.k3s.io | K3S_URL=https://192.168.1.100:6443 K3S_TOKEN="K10dabedfdbea3651aa97059622d5a4a378eacf36a9db4aa66d574bf798fbc3f693::server:bbee9726da23b1cad1d7a3fb6ce345a4" sh -

Paso 6: Verificación final (Ejecutar este comando en el nodo MASTER)
sudo kubectl get nodes
